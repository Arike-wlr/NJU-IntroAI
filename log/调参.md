import torch
import higher

class MetaLearner:
    def __init__(self, base_agent_class):
        self.base_agent_class = base_agent_class
        self.meta_optimizer = torch.optim.Adam(self.meta_parameters, lr=1e-3)
    
    def meta_train(self, tasks):
        """在多个任务上学习通用的超参数初始化"""
        for task_batch in tasks:
            # 内循环：在每个任务上快速适应
            task_losses = []
            for task in task_batch:
                agent = self.base_agent_class()
                # 用少量步骤在任务上训练
                adapted_agent = self.quick_adapt(agent, task, num_steps=5)
                task_loss = evaluate(adapted_agent, task)
                task_losses.append(task_loss)
            
            # 外循环：更新元参数
            meta_loss = torch.mean(torch.stack(task_losses))
            self.meta_optimizer.zero_grad()
            meta_loss.backward()
            self.meta_optimizer.step()

## 🧬 基于种群的方法 (Population-Based)

### 基于种群的训练 (PBT)
```python
class PopulationBasedTraining:
    def __init__(self, population_size=20):
        self.population = [self.initialize_agent() for _ in range(population_size)]
        self.hyperparameter_ranges = {
            'lr': (1e-5, 1e-2),
            'gamma': (0.8, 0.999),
            'epsilon_decay': (0.9, 0.9999)
        }
    
    def exploit_and_explore(self):
        """利用和探索阶段"""
        # 定期评估种群
        performances = [evaluate(agent) for agent in self.population]
        
        for i, agent in enumerate(self.population):
            if performances[i] < np.percentile(performances, 70):
                # 性能差的个体：从优秀个体复制参数并添加扰动
                best_agent = self.population[np.argmax(performances)]
                self.copy_and_perturb(agent, best_agent)
    
    def copy_and_perturb(self, target_agent, source_agent):
        """复制并扰动超参数"""
        # 复制网络参数
        target_agent.qnetwork_local.load_state_dict(source_agent.qnetwork_local.state_dict())
        
        # 扰动超参数
        for param_name, (low, high) in self.hyperparameter_ranges.items():
            current_value = getattr(source_agent, param_name)
            new_value = self.perturb_value(current_value, low, high)
            setattr(target_agent, param_name, new_value)
```

## 🔄 梯度-based超参数优化

### 通过梯度下降优化超参数
```python
class Hypergradient:
    def __init__(self, model, validation_data):
        self.model = model
        self.validation_data = validation_data
    
    def compute_hypergradient(self, hyperparams):
        """计算超参数的梯度"""
        # 需要可微分的训练过程
        with higher.innerloop_ctx(self.model, self.model.optimizer) as (fmodel, diffopt):
            # 内循环：用当前超参数训练
            for epoch in range(5):  # 少量训练步骤
                train_loss = self.compute_loss(fmodel, training_data)
                diffopt.step(train_loss)
            
            # 外循环：在验证集上计算损失
            validation_loss = self.compute_loss(fmodel, self.validation_data)
            
            # 计算超参数梯度
            hyper_grad = torch.autograd.grad(validation_loss, hyperparams.values())
            return hyper_grad
```

## 🌐 多保真度优化 (Multi-Fidelity)

### 连续减半 + 超带 (Hyperband)
```python
class HyperbandOptimizer:
    def __init__(self, max_budget=100, eta=3):
        self.max_budget = max_budget  # 最大训练预算（episodes）
        self.eta = eta  # 淘汰比例
    
    def run(self):
        """运行Hyperband算法"""
        s_max = int(np.log(self.max_budget) / np.log(self.eta))
        B = (s_max + 1) * self.max_budget
        
        for s in reversed(range(s_max + 1)):
            # 成功ive Halving
            n = int(np.ceil(B / self.max_budget / (s + 1) * self.eta ** s))
            r = self.max_budget * self.eta ** (-s)
            
            # 随机初始化n个配置
            configurations = [self.random_config() for _ in range(n)]
            
            for i in range(s + 1):
                # 用预算r * eta^i评估每个配置
                budget = r * self.eta ** i
                results = [self.evaluate_config(config, budget) for config in configurations]
                
                # 保留最好的1/eta比例的配置
                keep = max(1, int(len(configurations) / self.eta))
                configurations = [config for _, config in 
                                sorted(zip(results, configurations), key=lambda x: x[0])[-keep:]]
```

class JointArchitectureHyperparameterSearch:
    def __init__(self):
        self.search_space = {
            'architecture': {
                'num_layers': [2, 3, 4],
                'hidden_sizes': [[64, 64], [128, 64], [256, 128], [128, 128, 64]],
                'activation': ['relu', 'tanh', 'elu']
            },
            'hyperparameters': {
                'lr': (1e-5, 1e-2),
                'batch_size': [32, 64, 128, 256],
                'gamma': (0.9, 0.999)
            }
        }
    
    def search(self):
        """使用ENAS或DARTS等方法进行联合搜索"""
        # 定义可微分的架构参数
        architecture_weights = torch.nn.Parameter(torch.randn(len(self.search_space['architecture'])))
        
        # 联合优化架构权重和超参数
        optimizer = torch.optim.Adam([architecture_weights], lr=0.01)
        
        for step in range(100):
            # 采样架构
            arch_config = self.sample_architecture(architecture_weights)
            hyper_config = self.sample_hyperparameters()
            
            # 训练并评估
            performance = self.train_and_evaluate(arch_config, hyper_config)
            
            # 更新架构权重（使用REINFORCE或Gumbel-Softmax）
            loss = -performance  # 最大化性能
            loss.backward()
            optimizer.step()
```

class ZeroShotHyperparameterOptimization:
    """基于理论分析或元特征，无需训练即可预测好的超参数"""
    
    def predict_hyperparameters(self, environment_meta_features):
        """根据环境特征预测超参数"""
        # 元特征：状态维度、动作维度、episode长度等
        state_dim = environment_meta_features['state_dim']
        action_dim = environment_meta_features['action_dim']
        
        # 基于经验规则预测
        predicted_lr = min(1e-3, 1e-4 * state_dim / action_dim)
        predicted_batch_size = max(32, min(256, state_dim * 2))
        
        return {'lr': predicted_lr, 'batch_size': predicted_batch_size}
```

def next_level_tuning_strategy():
    """进阶调参策略"""
